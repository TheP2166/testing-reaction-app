    <script>
        const videoElement = document.getElementById('webcam');
        const canvasElement = document.getElementById('outputCanvas');
        const canvasCtx = canvasElement.getContext('2d');
        const promptInput = document.getElementById('promptInput');

        // Create a static fallback background image object
        let backgroundImg = new Image();
        backgroundImg.src = 'https://unsplash.com'; // Cyberpunk placeholder

        // Initialize MediaPipe Selfie Segmentation
        const selfieSegmentation = new SelfieSegmentation({
            locateFile: (file) => `https://jsdelivr.net{file}`
        });

        selfieSegmentation.setOptions({
            modelSelection: 1, // 1 for landscape/heavy background, 0 for closer portrait
        });

        selfieSegmentation.onResults(onResults);

        // Access user camera device
        async function startCamera() {
            try {
                const stream = await navigator.mediaDevices.getUserMedia({ 
                    video: { width: 1280, height: 720, facingMode: "user" } 
                });
                videoElement.srcObject = stream;
                videoElement.addEventListener('loadedmetadata', () => {
                    requestAnimationFrame(sendToProcessing);
                });
            } catch (err) {
                alert("Camera access denied or unavailable: " + err.message);
            }
        }

        async function sendToProcessing() {
            if (!videoElement.paused && !videoElement.ended) {
                await selfieSegmentation.send({ image: videoElement });
            }
            requestAnimationFrame(sendToProcessing);
        }

        // Composite frame processor: splits person from background and merges with AI layer
        function onResults(results) {
            canvasCtx.save();
            canvasCtx.clearRect(0, 0, canvasElement.width, canvasElement.height);

            // 1. Draw the custom background layer first
            if (backgroundImg.complete) {
                canvasCtx.drawImage(backgroundImg, 0, 0, canvasElement.width, canvasElement.height);
            } else {
                canvasCtx.fillStyle = '#222';
                canvasCtx.fillRect(0, 0, canvasElement.width, canvasElement.height);
            }

            // 2. Configure the mask overlay using the segmentation canvas
            canvasCtx.globalCompositeOperation = 'destination-out';
            canvasCtx.drawImage(results.segmentationMask, 0, 0, canvasElement.width, canvasElement.height);

            // 3. Draw the camera frame selectively only where the person is located
            canvasCtx.globalCompositeOperation = 'destination-over';
            canvasCtx.drawImage(results.image, 0, 0, canvasElement.width, canvasElement.height);

            canvasCtx.restore();
        }

        // Mock function connecting the frontend prompt to your AI Generation Engine
        async function updateBackground() {
            const prompt = promptInput.value;
            console.log("Generating background for prompt: " + prompt);
            
            // In production, you would fetch from your backend API router here
            // Example: const response = await fetch('/api/generate-bg', { method: 'POST', body: JSON.stringify({ prompt }) });
            // const data = await response.json();
            // backgroundImg.src = data.imageUrl;

            alert("Backend Pipeline Triggered:\nAI will now generate a '" + prompt + "' background canvas asset to slide behind your video feed.");
        }

        // Initialize webcam on window load
        window.onload = startCamera;
    </script>
